# PIC18F4580
- 40 Pin Low Power Microcontroller
- PORT A, B, C, D, E : 40 Pin (Port E is only 4 bit, rest are 8 bit)
- Internal Oscillator 4 Mhz, External 0-20 Mhz
- 10 bit//11 Channel ADC Converter 
- 3 x 16 bit Timer T0, T1, T3
- 1 x 8 bit Timer T2
- Program Memory 4K
- EPROM 256 bytes
- Power Supply 2 - 5.5v
- 10 bit PWM Module 
- CCP1, CCP2 Compare Module
- SPI, I2C, UART

## Port Programming

### Direction  
- 0 for Output and 1 for Input
- Entire Port : 
    - TRISC = 0x00; //Port C as output
    - TRISD = 0xff; //Port D as input

- Individual Bits : 
    - TRISCbits.RC0 = 0; //Bit 0 of Port C as Output
    - TRISCbits.RC1 = 1; //Bit 1 of Port C as Input

### Data Assignment
- 1 for High and 0 for Low
- Entire Port :
    - PORTC = 0xff; //Entire Port C set to High
    - PORTB = 0x00; //Entire Port B set to Low

- Individual Bits: 
    - PORTCbits.RC0 = 1 ; //Bit 0 of Port C set to High

```
    TRISCbits.RC1 = 1; //RW Output
    TRISCbits.RC2 = 1; //EN Output

    //ADC Register 
    //ADCON0 - A/D Enable bit, GO bit Enable
    /*
        b 00 0000 1 1   =  0x03

           Bit 7-6      00             Unimplemented
           Bit 5-2      0000           Channel 0 AN0/RA0
           Bit 1        1              A/D in progress
           Bit 0        1              A/D Enabled                   
    */
    ADCON0 = 0x03;

    //ADCON1 - Setting Pins as Analog or Digital
    /*
        AN0 selected in ADCON0 (Bit 5:2), so only AN0/RA0(Channel 0) needs to be set as analog.

        ADCON1 
        b 00 0 0 1110   = 0x0e
        
        Bit 7-6         00              Unimplemented
        Bit 5           0               +External Voltage Source
        Bit 4           0               -External Voltage Source
        Bit 3-0         1110            AN0 set as analog. AN10-AN1 digital
    */

    ADCON1 = 0x0e;

    //ADCON2 - justification, acquistion time, clock select
    /*

        ADCON2 
        b 1 0  001  000 = 0x88

        Bit 7           1               Right Justified (Little Endian)
        Bit 6           0               Unimplemented
        Bit 5:3         001             Acquistion Time. Selected 2 TAD or 0 TAD
        Bit 2:0         000             Clock Select : Fosc /2 = 4Mhz/2 = 2Mhz
    */

    ADCON2 = 0x88;

    TRISAbits.RA0 = 1; //Pot as Input
    init_LCD();

    while(1)
    {
        ADCONbits.GO = 1; //Set A/D conversion to in progress
        while(ADCON0bit.GO == 1); //Wait till conversion is complete

        adc_value = ADRESH;
        adc_value = adc_value << 8 + ADRESL
        
        /* 
            adc_value = (unsigned int)ADRESH << 8 + ADRESL
            
            Without unsigned int max will be 255 and not 1023
        */
        
        temp = adc_value;

        //Store ADC value in int arr per digit
        for(i = 3; i >= 0; i++)
        {
            arr[i] = temp%10;
            temp = temp/10;
        }
        

        //Display ADC value
            command(0x80); //Line 0 Position 1
            write("ADC Value : ");
            command(0xC0); //Line 2 Position 1
        for(i = 0; i < 4; i++)
        {
            data(arr[i]);
            delay(5);
        }

        command(0x01); //Clear screen
        delay(100);
    }

}
```

## Timer 0
```
/*
    Timer 0 

        1. Configure T0CON Register 
        2. Calculate TMR0H, TMR0L for desired Delay
        3. Rest Timer Interrupt Flag
        4. Start Timer 
        5. Wait for timer interrupt to overflow
        6. Stop Timer
        7. Reset Timer Interrupt Flag


        1. T0CON : b 0000 0111 -> 0x07
            Bit 7       0           Initially turned off. Enable Timer later.
            Bit 6       0           0 for 16 Bit Timer, 1 for 8 bit Timer
            Bit 5       0           0 for Internal Clock(Tosc), 1 for Timer 0 Clock
            Bit 4       0           1 for Rising Edge, 0 for Falling Edge
            Bit 3       0           1 -> No Pre-scaler, 0 -> Pre-scaler Assigned
            Bit 2-0     111         Pre-Scaler 1:256

        2. Calculate TMR0H, TMR0L

            F = Fosc/4 = 1 Mhz. ---- Internal Instruction Cycle is 4 
            F(timer) = F/Pre-scaler
                     = 1Mhz/256

            T(timer) = 1\F(timer) = 256 us

            100 ms Desired Delay

            TMR0H,TMROL = (2^16 -1) - [(Desired Delay)/T(timer)]
                        = 65535 - (100m)/(256u)
                        = 65144
                        = 0xfe78
                        
            TMR0H = 0xfe ,  TMR0L = 0X78

        
        3. INTCONbits.TMR0IF is the Interrupt Flag

        4. T0CONbits.TMR0ON is the Timer Enable Bit

        * Every 100ms Timer updates from 65144 to 65535 --- Timer 0,1,3
        * Every 100ms Timer updates from 0 to PR -- Timer 2

*/

#include<p18f4580.h>
#define LED PORTCbits.RC0

void delay(void);

int main()
{
    int i;

    TRISCbits.RC0 = 0; //LED Output

    T0CON = 0x07;

    while(1)
    {
        LED = 0;
        for(i = 0; i < 10; i++)
        {
            delay();
        }
        
        LED = 1;
        for(i = 0; i < 10; i++)
        {
            delay();
        }    
    }

    return 0;
}

void delay()
{
    TMR0H = 0xfe;
    TMR0L = 0x78;

    INTCONbits.TMR0IF = 0; //Reset Interrupt Flag
    T0CONbits.TMR0ON = 1; //Start Timer Timer

    while(INTCONbits.TMR0IF != 1); //Wait for overflow

    T0CONbits.TMR0ON = 0; //Stop Timer

    INTCONbits.TMR0IF = 0; //Reset Interupt Flag

}
```

## Timer 2
```
/*
    Timer 2 

        1. Configure T2CON Register 
        2. Calculate PR2 for desired Delay
        3. Rest Timer Interrupt Flag
        4. Start Timer 
        5. Wait for timer interrupt to overflow
        6. Stop Timer
        7. Reset Timer Interrupt Flag


        1. Configure T2CON

            //T2CON = 0x0b
            //b 0 0001 0 11 = 0000 1011 = 0x0b

            Bit 7       0       Unimplemented
            Bit 6-3     0001    1:2 Post Scale
            Bit 2       0       Timer ON bit. Enable it later
            Bit 1-0     11      16 Prescaler

        2. Calculate PR2

            //5ms Desired Delay
            Desired Delay =  (Pre-Scaler x PR2 x Post-Scaler)/ (Fosc/4)

            5 x 10^-3 = (16 x PR2 x 2)/ ((4x10^6)/4)

            PR =  156.25 = 156 = 0x9c

        3. PIR1bits.TMR2IF is Timer 2 Interrupt Flag

        4. T2CONbits.TMR2ON is the Timer Start Stop bit
*/

#include<p18f4580.h>
#define LED PORTCbits.RC0


void delay(void);

int main()
{
    int i;

    TRISCbits.RC0 = 0; //Output

    //Configured T2CON register
    T2CON = 0x0b;


    while(1)
    {
        LED = 0;
        //1 second total delay 
        for(i = 0; i < 200; i++)
        {
            delay();  
        }

        LED = 1;
        //1 second total delay 
        for(i = 0; i < 200; i++)
        {
            delay();
        }
    }
    return 0;
}


void delay()
{
    //PR2 for Desired Delay of 5 ms
    PR2 = 0x9c;

    PIR1bits.TMR2IF = 0; //Reset Timer

    T2CONbits.TMR2ON = 1; //Start Timer

    while(PIR1bits.TMR2IF != 1); //Wait for TMR2 - PR2 Match.
    
    T2CONbits.TMR2ON = 0; //Stop Timer

    PIR1bits.TMR2IF = 0; //Reset Timer

}
```

## UART TX
```
/*
UART :
    Asynchronous Communication based on baud rate
    Data Frame :
        IDLE(1)|START(0)|DATA(5-8)|PARITY(0 OR 1)|STOP(1)|IDLE(1)
        1 - HIGH, 0 - LOW

    UART Pins in PIC18F4580 : RC6/TX , RC7/RX

    Registers :
        TXSTA, RCSTA, SPBRG
    
    Buffers :
        TXREG, RCREG

    Register Configuration :

        TXSTA : b 0010 0100 = 0x24
            
            Bit 7       0           Clock Source (Ignore for UART)
            Bit 6       0           TX9. 1 for 9 bit Transmission, 0 for 8 bit Transmission
            Bit 5       1           TXEN. 1 for Transmit Enable
            Bit 4       0           SYNC. 1 for Sync mode, 0 for Async mode
            Bit 3       0           SENDB. 1 to send Sync Break, 0 for sync break tx completed
            Bit 2       1           BRGH. 1 for High Speed, 0 for Low Speed
            Bit 1       0           TRMT. 0 for Shift Register Full, 1 for Empty
            Bit 0       0           TX9D. Parity Bit

        RCSTA :  b 1001 0000 = 0x90
            Bit 7       1           Serial Port Enabled
            Bit 6       0           RX9. 1 for 9 bit , 0 for 8 bit reception
            Bit 5       0           SREN. Single Receive Enable bit. Ignore for ASYNC
            Bit 4       1           CREN. Enable Receiver
            Bit 3       0           ADEN. Ignore for 8 bit ASYNC mode
            Bit 2       0           FERR. Framming Error bit
            Bit 1       0           OERR. Overrun Error bit
            Bit 0       0           RX9D. 9th bit of received dat.

        SPBRG(N):
            BRGH selected as 1 (High Speed) in TCSTA bit 2
            Look up Table 18.1 ind data sheet for BRGH = 1 and ASYNC mode, to find Baud Rate formula

                Baud Rate = Fosc /(16(N+1)) 

                For Baud Rate = 9600, Fosc = 4Mhz

                    N = 25.0416 = 25

            SPBRG = 25 = 0x19
*/

#include<p18f4580.h>
#define TX PORTCbits.RC6
#define RX PORTCbits.RC7

void delay(void);

int main()
{
    int i;
    char arr[] = "Transmitting Message to UART\r\n";

    TRISCbits.RC6 = 0;
    TRISCbits.RC7 = 1;

    //UART Register Configuration
    TXSTA = 0x24;
    RCSTA = 0x90;
    SPBRG = 0x19;

    TXREG = 'a'; //Dummy value in Buffer

    while(1)
    {

        // Robust Error Checking
        // If an Overrun Error (OERR) occurs, CREN must be toggled to reset the receiver
        // If OERR is set, the receiver hardware literally "freezes" and will not receive any more data until you reset it.
        if (RCSTAbits.OERR)
        {
            RCSTAbits.CREN = 0; 
            RCSTAbits.CREN = 1; 
        }


        for(i = 0; arr[i] != '\0'; i++)
        {
            while(PIR1bits.TXIF == 0); //Wait for buffer to be empty. Will turn 1 when empty.
            TXREG = arr[i]; //Refill buffer

            // Read only bit. Hardware automatically clears it 
            //PIR1bits.TXIF = 0;  //Reset Transmit Interrupt Flag 
        }

        // Ensure the last character is transmitted before delaying
        while(TXSTAbits.TRMT == 0);

        delay();
    }
    return 0;
}

void delay()
{
    int i, j;
    for(i = 0; i < 100; i++){
        for(j = 0; j < 100; j++)
        {}
    }
}
```

```
            Bit 4      1        CREN. ASYNC Mode Enable Receiver

            Table 18.1 from datasheet for BRGH = 1 and ASYNC mode

            Desired Baud Rate = Fosc /(16(N + 1))
            
            For Baud Rate = 9600, Fosc = 4Mhz

            9600 = (4 x 10^6)/(16(N+1))

            N = 25.0416

            SPBRG = N = 25 = 0x19

        For Receiver, 

            Buffer RCREG

            PIR1bits.RCIF is the Receiver Interrupt Flag
                1 -> Buffer full
                0 -> Buffer Empty


    On Proteus : Connect PIC18F4580  TX/RC6 to UART Virtual Terminal RX 
                    and RX/RC7 to UART Virtual Terminal TX

                    Turn on simulation and on UART virtual terminal, right click

#define LED PORTCbits.RC0

int main()

    RX = 1;  //UART RX set as input

    //UART Register
    TXSTA = 0x24;
    RCSTA = 0X90;
    SPBRG = 0X19;

    while(1)
    {

        // Robust Error Checking
        // If an Overrun Error (OERR) occurs, CREN must be toggled to reset the receiver
        // If OERR is set, the receiver hardware literally "freezes" and will not receive any more data until you reset it.
        if (RCSTAbits.OERR)
        {

            RCSTAbits.CREN = 1;   //Enable Receiver

            case '1':
                LED = 1; //LED ON
                break;
            case '0':
                LED = 0; //LED OFF
                break;
        }

        //Read only. HW clears it
        //PIR1bits.RCIF = 0;
        delay();


    }
    return 0;
}
```

## Timer 0 Interrupt
```
/*Timer as an interrupt

1. Enable IPEN from RCON register
2. Setup INTCON Register to enable Global and Peripheral Interrupt
3. Enable Interrupt Overflow Enable Bit from INTCON, PIE register for Timer 0/1/2
4. Don't turn on the timer yet
5. Setup T0CON/T1CON/T2CON register
6. Calculate Delay (TMR0H,TMR0L)/(TMR1H, TMR1L)/PR2
7. Write Interrupt Vector and ISR function.
8. Enable/Start the timer in main
9. Add the (TMR0H,TMR0L)/(TMR1H, TMR1L)/PR2 in ISR
10. Toggle the LED in ISR and reset the Interrupt Flag bit
*/

#include<p18f4580.h>
#define LED PORTCbits.RC0

void Timer_0(void);

#pragma code InterruptVectorHigh=0x08
    void InterruptVectorHigh(void){
        _asm goto Timer_0 _endasm
    }
#pragma code

#pragma interrupt Timer_0
void Timer_0(void){
    
    TMR0H = 0xf8;
    TMR0L = 0x5d;
    LED = ~LED;  //Toggle LED
    INTCONbits.TMR0IF = 0; //Reset Interrupt Flag
}


int main(){

    TRISCbits.RC0 = 0; //Output
    
    //Enable Interrupt Priority
    RCONbits.IPEN = 1;

    /*
    Enable Global Interrupt, Peripheral Interrupt
        b 1110 0000 = 0xe0

        Bit 7   1   Global Interrupt High Priority Interrupt Enabled
        Bit 6   1   Enable Low Priority Peripheral Interrupt
        Bit 5   1   Enable TMR0 Interrupt
        Bit 4   0   Disable INT0 External Interrupt
        Bit 3   0   Disable RB Port change
        Bit 2   0   TMR0IF. No overflow. Flag unset.
        Bit 1   0   INT0 unused
        Bit 0   0   RB0 port change unused
    */

    INTCON = 0xe0;

    /*
        Configure T0CON register
        Bit 7       0       TMR0ON. Stop Timer 0
        Bit 6       0       0 for 16 bit timer
        Bit 5       0       Internal Clock
        Bit 4       0       Edge select low to high
        Bit 3       0       PSA Assigned
        Bit 2:0     111     Pre-scaler 256

        b 0000 0111 = 0x07
    */

    T0CON = 0x07;


    /*
    Desired Delay 500ms
    
        F = Fosc/4 = 1 Mhz

        F(timer) = F / Pre-scaler = 1/256 Mhz 

        T(timer) = 256 us

        TMR0H,TMR0L = (2^16 - 1) - (Desired Delay/T(timer))
                    = 65535 - (500 x 10^-3)/(256 x 10^-6)
                    = 63581
                    = 0xf85d
    */

    TMR0H = 0xf8;
    TMR0L = 0x5d;

    LED = 0; //Turn off LED
    T0CONbits.TMR0ON = 1; //Enable Timer

    while(1){}

    return 0;
}
```

```
/*Timer as an interrupt

8. Enable/Start the timer in main
9. Add the (TMR0H,TMR0L)/(TMR1H, TMR1L)/PR2 in ISR
10. Toggle the LED in ISR and reset the Interrupt Flag bit

void Timer_1(void);

#pragma code InterruptVectorHigh=0x08
    void InterruptVectorHigh(void){

        _asm goto Timer_1 _endasm

    }
#pragma code

#pragma interrupt Timer_1
void Timer_1(void){

        TMR1H = 0x0b;
        TMR1L = 0xdb;

        LED = ~LED;

        PIR1bits.TMR1IF = 0; //Reset Timer 1 Interrupt Flag
}

int main()
{

    TRISCbits.RC0 = 0; //LED as output

    //Enable Interrupt Priority
    RCONbits.IPEN = 1;

    // Explicitly set Timer1 to High Priority since IPEN=1
    IPR1bits.TMR1IP = 1;

    
    //Enable Global Interrupt, Peripheral Interrupt
    //b 1100 0000 = 0xc0
    INTCON  = 0xc0;

    //Enable Timer 1 Overflow Interrupt Bit
    PIE1bits.TMR1IE = 1;

    //LED off
    LED = 0;

    //Enable T1CON Register
    //b 1 0 11 0 0 0 0 = 1011 0000 = 0xb0
    T1CON = 0xb0; //16 bit mode, Pre-scaler  of 8, Stop Timer1

    //Delay Timer
    /*
        F = Fosc/4 =. 1 Mhz
        F(timer) = F / Pre-scaler = 1 Mhz/ 8 
        T(timer) = 1/F(Timer) = 8 us

        Desired Delay 500ms

        TMR1H, TMR1L = (2^16 - 1) - [(Desired Delay)/T(timer)]
                     = 65535 - [(500 x 10^-3)/(8 x 10^-6)]
                     = 3035
                     = 0x0bdb

        TMR1H = 0x0b | TMR1L = 0xdb

    */
    TMR1H = 0x0b;
    TMR1L = 0xdb;
    PIR1bits.TMR1IF = 0; //Reset Interrupt Flag
    T1CONbits.TMR1ON = 1; //Enable Timer

    while(1){}
    return 0;
}
```
